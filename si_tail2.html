############################################################
# Z expression
############################################################

function compute_Z(x1,y1,x2,y2,t)

    Δx = x1 - x2
    Δy = y1 - y2
    Δ  = Δx + Δy

    # if abs(Δ) < 1e-12
    #    return 0.0
    # end

    return (Δx*Δy/(2*Δ))*tanh(0.5*t*Δ)

end

############################################################
# Simulation
############################################################

function simulate_Z(t)

    Z = Float64[]

    for i in 1:N

        Δx = x1 - x2

        Δ  = Δx + Δy

        z = compute_Z(x1,y1,x2,y2,t)

        if isfinite(z)

        end

    end

    sort!(Z)

    return Z

end
