using Distributions
using QuadGK
using Random
using Statistics
using Plots

############################################################
# Parameters
############################################################

ρ = -0.5              # correlation of bivariate normal
σ = 0.1             # standard deviation
p = 0.5              # mixture weight
N = 2000_000          # number of simulations

a = 1.0             # dissociation rate
t = 500.0             # integration upper limit

Random.seed!(1)

############################################################
# Bivariate normal distribution
############################################################

Σ = [σ^2  ρ*σ^2;
     ρ*σ^2  σ^2]

dist = MvNormal([-0.01,-0.01], Σ)

############################################################
# Integral appearing in Z
############################################################

function kernel_integral(X1,Y1,X2,Y2,a,t,p)

    A = X1 + Y1
    B = X2 + Y2

    f(u) = begin

        num = a*exp(-a*u)*p*(1-p)*exp((A+B)*u)

        denom = (p*exp(A*u) + (1-p)*exp(B*u))^2

        num/denom

    end

    # if t>1
    #     val,err = quadgk(f,t-1,t)
    # else
    #     val,err = 0,0
    # end

    val,err = quadgk(f,0,t)

    return val

end

############################################################
# Single draw of Z
############################################################

function draw_Z(a,t,p)

    X1,Y1 = rand(dist)
    X2,Y2 = rand(dist)

    I = kernel_integral(X1,Y1,X2,Y2,a,t,p)

    return -(X2-X1)*(Y2-Y1)*I

end

############################################################
# Simulation
############################################################

Z = Vector{Float64}(undef,N)

for i in 1:N
    Z[i] = draw_Z(a,t,p)
end
